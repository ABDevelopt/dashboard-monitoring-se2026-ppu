// Inisialisasi sinkronisasi data progres survei
import { TakaCharts, watchTheme } from '@bps/taka-ui';

async function fetchSurveyProgress(surveyId = 'SE2026') {
  const endpoint = `/api/progress/${surveyId}`;
  const res = await fetch(endpoint);
  const data = await res.json();
  
  if (data.status === 200 && data.active === true) {
    return {
      realisasi: 9010,
      target: 12840,
      persentase: 70.1
    };
  }
}
query-agregat.sql
SQL 
-- Rekapitulasi agregat per kecamatan untuk PPU
SELECT 
    kecamatan,
    COUNT(sls_id) AS total_sls,
    SUM(realisasi_fasih) AS capaian,
    ROUND(AVG(pct), 2) AS rata_persen
FROM monitoring_progress
WHERE kode_kabupaten = '6409' 
  AND status_aktif = true
GROUP BY kecamatan
ORDER BY capaian DESC;
